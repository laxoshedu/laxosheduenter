    <script>
        // --- THEME ENGINE ---
        const body = document.body;
        const themeBtn = document.querySelector('.theme-toggle-btn');
        const themeIcon = document.querySelector('.theme-icon');

        const savedTheme = localStorage.getItem('portalTheme') || 'dark';
        body.setAttribute('data-theme', savedTheme);
        updateIcon(savedTheme);

        themeBtn.addEventListener('click', () => {
            const currentTheme = body.getAttribute('data-theme');
            const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
            
            body.setAttribute('data-theme', newTheme);
            localStorage.setItem('portalTheme', newTheme);
            updateIcon(newTheme);
        });

        function updateIcon(theme) {
            themeIcon.innerText = theme === 'dark' ? '☀️' : '🌙';
        }

        // --- ENTRANCE ANIMATION ---
        document.addEventListener("DOMContentLoaded", () => {
            const target = document.querySelector('.animate-target');
            setTimeout(() => {
                target.classList.add('animate-show');
            }, 100);
        });
    </script>
